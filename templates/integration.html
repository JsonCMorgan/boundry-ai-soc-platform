Integration Setup

Connect your application to Boundry.AI in under an hour.

📋 How it works: Add a small logging function to your app. Every time a security-relevant event happens (a login attempt, a search, a failed auth), your app sends a single POST request to Boundry.AI. The agent analyses those events and generates incident reports automatically — you don't need to change anything else.

Step 2 — What to Monitor

Add monitoring calls to these points in your application. You don't need all of them on day one — start with login and search.

  • ▸
    Login route — failed attempt
    Every time someone enters the wrong password. This is how we detect brute force and credential stuffing.
  • ▸
    Login route — successful login
    Establishes a baseline of normal logins so we can flag suspicious ones (new IP, unusual hours).
  • ▸
    Search / query routes
    Pass the search query in extra — we scan it for SQL injection patterns.
  • ▸
    Admin / restricted routes (403 responses)
    Log when a user is denied access to a route they shouldn't reach — catches privilege escalation attempts.
  • ▸
    Registration / new account creation
    Useful for detecting mass account creation or enumeration attempts.
  • ▸
    Form inputs / comment fields
    Pass the submitted value in extra — we detect XSS payloads and script injection.

# boundry.py — drop this file into your project root import threading, requests BOUNDRY_API_KEY = "YOUR_API_KEY" BOUNDRY_ENDPOINT = "{{ ingest_url }}" def log_event(event_type, username="", ip="", extra=""): """Fire-and-forget — runs in a background thread so it never slows your app.""" def _send(): try: requests.post( BOUNDRY_ENDPOINT, json={"event_type": event_type, "username": username, "ip": ip, "extra": extra}, headers={"X-API-Key": BOUNDRY_API_KEY}, timeout=3, ) except Exception: pass # never block your users if monitoring is down threading.Thread(target=_send, daemon=True).start() # ── In your Flask login route ───────────────────────────────────────────── from boundry import log_event @app.route("/login", methods=["POST"]) def login(): username = request.form.get("username", "") ip = request.remote_addr user = get_user(username) if not user: log_event("ACCOUNT_ENUM", username=username, ip=ip) # username doesn't exist flash("Invalid credentials") return redirect("/login") if not check_password(user, request.form.get("password")): log_event("LOGIN_FAILED", username=username, ip=ip) flash("Invalid credentials") return redirect("/login") log_event("LOGIN_SUCCESS", username=username, ip=ip) session["user"] = username return redirect("/") # ── In your search route ────────────────────────────────────────────────── @app.route("/search") def search(): q = request.args.get("q", "") log_event("SEARCH", username=session.get("user"), ip=request.remote_addr, extra=q) # ... rest of your search logic # ── On a 403 / unauthorised access ─────────────────────────────────────── @app.errorhandler(403) def forbidden(e): log_event("PRIV_ESC_ATTEMPT", username=session.get("user"), ip=request.remote_addr, extra=request.path) return "Forbidden", 403
# boundry.py — drop this in your Django project root (next to manage.py) import threading, requests BOUNDRY_API_KEY = "YOUR_API_KEY" BOUNDRY_ENDPOINT = "{{ ingest_url }}" def log_event(event_type, username="", ip="", extra=""): def _send(): try: requests.post( BOUNDRY_ENDPOINT, json={"event_type": event_type, "username": username, "ip": ip, "extra": extra}, headers={"X-API-Key": BOUNDRY_API_KEY}, timeout=3, ) except Exception: pass threading.Thread(target=_send, daemon=True).start() # ── In your Django login view ───────────────────────────────────────────── from django.contrib.auth import authenticate, login from boundry import log_event def login_view(request): if request.method == "POST": username = request.POST.get("username", "") ip = request.META.get("REMOTE_ADDR", "") user = authenticate(request, username=username, password=request.POST.get("password")) if user: login(request, user) log_event("LOGIN_SUCCESS", username=username, ip=ip) return redirect("/") else: log_event("LOGIN_FAILED", username=username, ip=ip) return render(request, "login.html") # ── Middleware to catch 403s automatically (settings.py → MIDDLEWARE list) ─ # boundry_middleware.py class BoundryMiddleware: def __init__(self, get_response): self.get_response = get_response def __call__(self, request): response = self.get_response(request) if response.status_code == 403: from boundry import log_event username = request.user.username if request.user.is_authenticated else "" log_event("PRIV_ESC_ATTEMPT", username=username, ip=request.META.get("REMOTE_ADDR", ""), extra=request.path) return response
// boundry.js — drop this in your project root const BOUNDRY_API_KEY = 'YOUR_API_KEY'; const BOUNDRY_ENDPOINT = '{{ ingest_url }}'; function logEvent(eventType, username = '', ip = '', extra = '') { // Non-blocking — errors are swallowed so monitoring never breaks your app fetch(BOUNDRY_ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json', 'X-API-Key': BOUNDRY_API_KEY }, body: JSON.stringify({ event_type: eventType, username, ip, extra }), signal: AbortSignal.timeout(3000), }).catch(() => {}); // fire and forget } module.exports = { logEvent }; // ── In your Express login route ─────────────────────────────────────────── const { logEvent } = require('./boundry'); app.post('/login', async (req, res) => { const { username, password } = req.body; const ip = req.ip; const user = await User.findOne({ username }); if (!user || !await user.checkPassword(password)) { logEvent('LOGIN_FAILED', username, ip); return res.status(401).send('Invalid credentials'); } logEvent('LOGIN_SUCCESS', username, ip); req.session.user = username; res.redirect('/'); }); // ── Express middleware: catch 403s automatically ─────────────────────────── app.use((req, res, next) => { const originalSend = res.send.bind(res); res.send = (body) => { if (res.statusCode === 403) { logEvent('PRIV_ESC_ATTEMPT', req.session?.user || '', req.ip, req.path); } return originalSend(body); }; next(); });
// boundry.php — drop this in your theme or plugin folder <?php define('BOUNDRY_API_KEY', 'YOUR_API_KEY'); define('BOUNDRY_ENDPOINT', '{{ ingest_url }}'); function boundry_log_event($event_type, $username = '', $ip = '', $extra = '') { // Non-blocking post using wp_remote_post (WordPress) or curl fallback $body = json_encode([ 'event_type' => $event_type, 'username' => $username, 'ip' => $ip, 'extra' => $extra, ]); if (function_exists('wp_remote_post')) { wp_remote_post(BOUNDRY_ENDPOINT, [ 'body' => $body, 'headers' => ['Content-Type' => 'application/json', 'X-API-Key' => BOUNDRY_API_KEY], 'timeout' => 3, 'blocking' => false, // non-blocking ]); } else { $ch = curl_init(BOUNDRY_ENDPOINT); curl_setopt_array($ch, [ CURLOPT_POST => true, CURLOPT_POSTFIELDS => $body, CURLOPT_HTTPHEADER => ['Content-Type: application/json', 'X-API-Key: ' . BOUNDRY_API_KEY], CURLOPT_TIMEOUT => 3, CURLOPT_RETURNTRANSFER => true, ]); curl_exec($ch); curl_close($ch); } } // ── WordPress: hook into failed and successful logins ───────────────────── add_action('wp_login_failed', function($username) { boundry_log_event('LOGIN_FAILED', $username, $_SERVER['REMOTE_ADDR']); }); add_action('wp_login', function($username) { boundry_log_event('LOGIN_SUCCESS', $username, $_SERVER['REMOTE_ADDR']); }); // ── Catch requests to wp-admin by non-admins ────────────────────────────── add_action('admin_init', function() { if (!current_user_can('manage_options')) { $user = wp_get_current_user(); boundry_log_event('PRIV_ESC_ATTEMPT', $user->user_login, $_SERVER['REMOTE_ADDR'], $_SERVER['REQUEST_URI']); } });
